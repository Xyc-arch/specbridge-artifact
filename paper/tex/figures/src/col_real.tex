\newcommand{\modelbars}[2]{% #1 csv file, #2 model index 0..3

  \noexpand\addplot[fill=white, draw=\mc, line width=0.7pt, bar shift=-6.2pt, restrict x to domain=#2:#2, sb yerr, error bars/error bar style={draw=sbInk}]
    table[x=x, y=o, y error minus=om, y error plus=op] {#1};
  \noexpand\addplot[fill=\mc!38, draw=\mc!38, line width=0.7pt, bar shift=0pt, restrict x to domain=#2:#2, sb yerr, error bars/error bar style={draw=sbInk}]
    table[x=x, y=c, y error minus=cm, y error plus=cp] {#1};
  \noexpand\addplot[fill=\mc, draw=\mc, line width=0.7pt, bar shift=6.2pt, restrict x to domain=#2:#2, sb yerr, error bars/error bar style={draw=sbInk}]
